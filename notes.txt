@NotNull -> Não aceita valores nulos
@NotEmpty -> Não aceita uma string vazia ""
@NotBlank -> Não aceita uma string em branco "    "

@DecimalMin("0") -> Valor mínimo a ser aceito (deve ser uma string)
@PositiveOrZero -> Só aceita zero ou positívo
